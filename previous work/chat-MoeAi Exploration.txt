```javascript
// ─── RAG SEARCH (Clean, Single Instance) ──────────────────────────────────────
async function searchRAG(query) {
  try {
    // 1. Embed query
    const embRes = await fetch('http://127.0.0.1:5002/embed', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ texts: [query] })
    });
    if (!embRes.ok) return '';
    const embData = await embRes.json();
    
    // 2. Search Supabase
    const {  chunks, error } = await supabase.rpc('search_knowledge', {
      query_embedding: embData.embeddings[0],
      match_count: 3,
      filter_subject: null
    });
    
    if (error || !chunks || chunks.length === 0) return '';
    
    // 3. Format Context
    return '\n--- RELEVANT LECTURE CONTEXT ---\n' + 
      chunks.map(c => `[${c.subject}]: ${c.content}`).join('\n') + 
      '\n--- END CONTEXT ---\n';
  } catch (err) {
    return ''; // Fail silently if RAG is offline
  }
}

// ─── AI PROVIDERS & FALLBACK CHAIN ────────────────────────────────────────────
async function callAI(prompt, messages) {
  const msgs = [{ role: 'system', content: prompt }, ...messages];

  // 1. KoboldCpp (Primary)
  if (KOBOLD_URL) {
    try {
      const res = await fetch(`${KOBOLD_URL}/api/v1/generate`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          prompt: prompt + '\n' + messages.map(m => `${m.role}: ${m.content}`).join('\n') + '\nAssistant:',
          max_length: MAX_TOKENS, temperature: 0.7,
          stop_sequence: ['User:', 'Human:']
        })
      });
      const data = await res.json();
      if (data.results?.[0]?.text) return data.results[0].text.trim();
    } catch (e) { console.log('[WARN] Kobold failed, falling back...'); }
  }

  // 2. Groq (Fallback)
  if (GROQ_KEY) {
    try {
      const res = await fetch('https://api.groq.com/openai/v1/chat/completions', {
        method: 'POST',
        headers: { 'Authorization': `Bearer ${GROQ_KEY}`, 'Content-Type': 'application/json' },
        body: JSON.stringify({ model: 'llama-3.3-70b-versatile', messages: msgs, temperature: 0.7 })
      });
      const data = await res.json();
      if (data.choices?.[0]?.message?.content) return data.choices[0].message.content;
    } catch (e) { console.log('[WARN] Groq failed, falling back...'); }
  }

  // 3. Gemini (Last Resort)
  if (GEMINI_KEY) {
    try {
      const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${GEMINI_MODEL}:generateContent?key=${GEMINI_KEY}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          systemInstruction: { parts: [{ text: prompt }] },
          contents: messages.map(m => ({ role: m.role === 'assistant' ? 'model' : 'user', parts: [{ text: m.content }] })),
          generationConfig: { temperature: 0.7 }
        })
      });
      const data = await res.json();
      if (data.candidates?.[0]?.content?.parts?.[0]?.text) return data.candidates[0].content.parts[0].text;
    } catch (e) { console.log('[ERROR] Gemini failed'); }
  }
  
  return "⚠️ All AI brains are offline or busy. Try again in a moment.";
}

// ─── MAIN CHAT HANDLER ────────────────────────────────────────────────────────
export async function chatWithMoeAI(userId, message) {
  // 1. Get Student Memory & History
  const {  memory } = await supabase.from('student_memory').select('*').eq('student_id', userId).single();
  const { data: history } = await supabase.from('messages').select('role, content').eq('student_id', userId).order('created_at', { ascending: false }).limit(10);

  // 2. RAG Search
  const context = await searchRAG(message);

  // 3. Build System Prompt
  const systemPrompt = `You are MoeAI, the smartest senior student for CS at FUE.
${OMEGA_RULES}
${FEW_SHOT_EXAMPLES}
Student Memory: ${JSON.stringify(memory || {})}
${context ? context : ''}
`;

  // 4. Generate Response
  const reply = await callAI(systemPrompt, (history || []).reverse());

  // 5. Save to DB
  await supabase.from('messages').insert([
    { student_id: userId, role: 'user', content: message },
    { student_id: userId, role: 'agent', content: reply, agent_id: 'aria' }
  ]);

  // 6. Trigger Auto-Learn (Optional)
  // if (typeof triggerAutoLearn === 'function') triggerAutoLearn(userId, message, reply);

  return reply;
}
```

#### 5. `simple-ingest.js` (For your RAG Lectures)
```javascript
import fs from 'fs';
import path from 'path';
import fetch from 'node-fetch';
import { supabase } from './supabase.js';

const DATA_DIR = '../data/deepseek_lectures';
const EMBED_URL = 'http://127.0.0.1:5002/embed';

async function ingest() {
  if (!fs.existsSync(DATA_DIR)) return console.log("No data folder found.");
  
  const files = fs.readdirSync(DATA_DIR).filter(f => f.endsWith('.txt'));
  for (const file of files) {
    const text = fs.readFileSync(path.join(DATA_DIR, file), 'utf8');
    const chunks = text.split('\n\n').filter(c => c.length > 50);
    const subject = file.replace('.txt', '').toUpperCase();
    
    console.log(`Ingesting ${file} (${chunks.length} chunks)...`);
    
    for (const chunk of chunks) {
      const embRes = await fetch(EMBED_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ texts: [chunk] })
      });
      const embData = await embRes.json();
      
      await supabase.from('knowledge_chunks').insert({
        content: chunk,
        embedding: embData.embeddings[0],
        subject: subject,
        source: file
      });
    }
  }
  console.log("✅ Ingestion Complete.");
}

ingest();
```

```python
# ~/moeai_collective/test.py
import os, sqlite3, asyncio
from telegram import Update
from telegram.ext import Application, MessageHandler, filters, ContextTypes
from groq import Groq

# Load config
TELEGRAM_TOKEN = os.getenv('TELEGRAM_TOKEN')
GROQ_KEYS = [os.getenv(f'GROQ_KEY_{i}') for i in range(1,5) if os.getenv(f'GROQ_KEY_{i}')]
DB_PATH = os.path.expanduser('~/moeai_collective/memory.db')

# Simple Groq caller with key rotation
def call_groq(prompt: str, key_index: int = 0) -> str:
    client = Groq(api_key=GROQ_KEYS[key_index % len(GROQ_KEYS)])
    response = client.chat.completions.create(
        model="llama-3.3-70b-versatile",
        messages=[{"role": "user", "content": prompt}],
        temperature=0.85,
        max_tokens=1024
    )
    return response.choices[0].message.content

# Load agent prompt from memory
def get_agent_prompt(agent: str) -> str:
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()
    cursor.execute("SELECT system_prompt FROM agent_memories WHERE agent=?", (agent,))
    result = cursor.fetchone()
    conn.close()
    return result[0] if result else "You are a helpful assistant."

# Save conversation to memory
def log_message(sender: str, recipient: str, content: str, tags: str = ""):
    conn = sqlite3.connect(DB_PATH)
    conn.execute(
        "INSERT INTO conversations (sender, recipient, content, context_tags) VALUES (?, ?, ?, ?)",
        (sender, recipient, content, tags)
    )
    conn.commit()
    conn.close()

# Simple router: decide which agent should respond
def route_message(text: str) -> str:
    text_lower = text.lower()
    if "@jarvis" in text_lower or "sir" in text_lower: return "jarvis"
    if "@sentinel" in text_lower or "scrape" in text_lower or "schedule" in text_lower: return "sentinel"
    if "@nova" in text_lower or "advanced" in text_lower or "deep" in text_lower: return "nova"
    return "aria"  # default

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text
    sender = update.message.from_user.username or "unknown"
    
    # Skip if message is from bot itself
    if update.message.from_user.is_bot: return
    
    # Determine which agent responds
    agent = route_message(text)
    prompt = f"{get_agent_prompt(agent)}\n\nGroup chat context:\n{text}\n\nRespond as {agent.upper()}:"
    
    # Get response from Groq
    reply = call_groq(prompt)
    
    # Log to memory
    log_message(sender, agent, text)
    log_message(agent, "group", reply)
    
    # Send reply to group
    await update.message.reply_text(f"*{agent.upper()}:* {reply}", parse_mode='Markdown')

# Run
app = Application.builder().token(TELEGRAM_TOKEN).build()
app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))
print("🤖 MoeAI Collective bot starting...")
app.run_polling()
```

```javascript
// ─── PROVIDER: GROQ VISION (Primary for Images) ───────────────────────────────
async function callGroqVision(systemPrompt, userText, imageBase64, temperature = 0.85) {
  if (!GROQ_API_KEY) throw new Error('No Groq key configured');
  
  const content = [];
  if (userText) content.push({ type: "text", text: userText });
  if (imageBase64) content.push({ 
    type: "image_url", 
    image_url: { url: imageBase64, detail: "high" } 
  });

  const res = await fetch('https://api.groq.com/openai/v1/chat/completions', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${GROQ_API_KEY}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      model: 'llama-3.2-90b-vision-preview', // 90B Vision Model
      messages: [
        { role: 'system', content: systemPrompt },
        { role: 'user', content: content }
      ],
      temperature,
      max_tokens: MAX_TOKENS,
    }),
  });

  if (!res.ok) throw new Error(`Groq Vision error: ${res.status} ${await res.text()}`);
  const data = await res.json();
  const text = data.choices?.[0]?.message?.content;
  if (!text) throw new Error('Empty Groq Vision response');
  return { text, tokensUsed: data.usage?.total_tokens || 0, provider: 'groq-vision' };
}
```

```javascript
import fetch from 'node-fetch';

async function telegramPhotoToBase64(fileId, botToken) {
  try {
    // 1. Get file path from Telegram
    const fileInfoRes = await fetch(`https://api.telegram.org/bot${botToken}/getFile?file_id=${fileId}`);
    const fileInfo = await fileInfoRes.json();
    if (!fileInfo.ok) return null;
    
    const fileUrl = `https://api.telegram.org/file/bot${botToken}/${fileInfo.result.file_path}`;
    
    // 2. Download image and convert to base64
    const imgRes = await fetch(fileUrl);
    const buffer = await imgRes.arrayBuffer();
    const base64 = `data:${imgRes.headers.get('content-type') || 'image/jpeg'};base64,${Buffer.from(buffer).toString('base64')}`;
    return base64;
  } catch (e) {
    console.error('❌ Failed to process image:', e.message);
    return null;
  }
}
```
